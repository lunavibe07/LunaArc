cmake_minimum_required(VERSION 3.20)
project(LunaArcContextMenu LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "The Explorer extension is Windows-only")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(LunaArc.ContextMenu SHARED ContextMenu.cpp ContextMenu.def)
target_compile_definitions(LunaArc.ContextMenu PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(LunaArc.ContextMenu PRIVATE shell32 shlwapi ole32 uuid)
if(MSVC)
  target_compile_options(LunaArc.ContextMenu PRIVATE /W4 /WX /EHsc /utf-8)
endif()
add_executable(ContextMenuChecks ContextMenuChecks.cpp)
if(MSVC)
  target_compile_options(ContextMenuChecks PRIVATE /W4 /WX /EHsc /utf-8)
endif()
target_compile_definitions(ContextMenuChecks PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(ContextMenuChecks PRIVATE shell32 shlwapi ole32 uuid)
enable_testing()
add_test(NAME ExplorerCommandContract COMMAND ContextMenuChecks $<TARGET_FILE:LunaArc.ContextMenu>)
